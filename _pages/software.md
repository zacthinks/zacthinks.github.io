<article class="project-entry" id="geco">

    <div class="project-entry__title">GeCo — Geometric Coder</div>
    <div class="project-entry__links">

<article class="project-entry" id="lipe">

    <div class="project-entry__title">LIPE — Latent Interview Protocol Engineer</div>
    <div class="project-entry__links">
      <a class="project-icon-link" href="https://github.com/zacthinks/lipe" title="LIPE on GitHub" aria-label="LIPE on GitHub"><i class="fab fa-github" aria-hidden="true"></i></a>

  </div>
